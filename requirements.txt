# Trace Engine 1 dependencies.

# Reconstruction (Engine 1b) — calls claude-haiku-4-5.
# Stage 4 (Managed Agents) needs the agent surface (client.beta.agents/sessions);
# if "Compile & test agents" reports no Managed Agents API, run: pip install -U anthropic
anthropic>=0.40.0

# Local capture (Engine 1a, macOS only). Not needed to run reconstruction or
# the test suite; install on the capture machine.
pynput>=1.7.6 ; sys_platform == "darwin"

# The macOS capture spine has depended on these since window capture moved to
# CGWindowList, and Accessibility capture (trace/capture/ax.py) needs
# ApplicationServices for AXUIElement — but neither was ever declared. That gap
# fails the worst possible way: every import is inside a try/except so capture
# degrades silently to "no events" rather than erroring, which is indistinguishable
# from a quiet workday. Declared now so a fresh machine or a bundled build cannot
# ship without them.
#   Quartz              -> CGWindowList (window titles), IOHIDCheckAccess
#   ApplicationServices -> AXUIElement* (focused element, click targets, AX titles)
pyobjc-framework-Quartz>=9.0 ; sys_platform == "darwin"
pyobjc-framework-ApplicationServices>=9.0 ; sys_platform == "darwin"

# Local test UI (trace.ui) — secure API-key storage in the macOS Keychain.
# Optional: without it, the UI falls back to a 0600 config file. The UI backend
# itself uses only the standard library (http.server).
keyring>=24.0 ; sys_platform == "darwin"

# The collector and reconstruction use only the Python standard library
# otherwise (http.server, json, fcntl, argparse, unittest).

# Stage 4 (Agent Compiler) — emit/validate Managed Agents agent-definition YAML.
PyYAML>=6.0

# Phase 1 (cloud backend). Both cross-platform, so unconditional:
# - python-dotenv: load a git-ignored .env on a local run (no-op in the cloud,
#   where Render supplies the same names as real environment variables).
# - psycopg (v3): connect to the managed Postgres via DATABASE_URL and create the
#   users/sessions schema on startup. Imported lazily, so a DB-less local run
#   never needs it at runtime; pinned here so the container build has it.
python-dotenv>=1.0
psycopg[binary]>=3.1

# CA bundle for the stdlib Pipedream Connect HTTPS calls (connect_client.py), so TLS
# verification doesn't depend on the OS cert store. Also pulled in transitively by anthropic.
certifi>=2024.0

# Phase 2 (Supabase backend). The supabase-py client wraps Auth, Postgres (PostgREST),
# and Storage behind a single client object. Imported lazily so a fully-local run
# never needs it at runtime.
supabase>=2.0
